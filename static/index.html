sun times

						

							Returns sunrise, sunset, solar noon, day length, and twilight
							times for a coordinate and calendar date. All timestamps are in
							UTC.
						

						
GET /json?lat=40.7128&lng=-74.0060&date=2026-06-12

					


					

						
query parameters

						

							
								

									name
									required
									description
								

							
							
								

									lat
									yes
									Latitude from -90 to 90.
								

								

									lng
									yes
									Longitude from -180 to 180.
								

								

									date
									no
									
										A calendar date in YYYY-MM-DD format, or
										today. Defaults to the current UTC date.
									
								

							
						

					


					

						
examples

						
curl "/json?lat=40.7128&lng=-74.0060&date=today"

						
const response = await fetch(
  "/json?lat=40.7128&lng=-74.0060&date=2026-06-12"
);
const data = await response.json();

					


					

						
response

						
{
  "results": {
    "sunrise": "2026-06-12T13:04:30+00:00",
    "sunset": "2026-06-12T04:07:41+00:00",
    "solar_noon": "2026-06-12T20:36:06+00:00",
    "day_length": 54190.730342635026,
    "civil_twilight_begin": "2026-06-12T12:31:15+00:00",
    "civil_twilight_end": "2026-06-12T04:40:57+00:00",
    "nautical_twilight_begin": "2026-06-12T11:48:53+00:00",
    "nautical_twilight_end": "2026-06-12T05:23:18+00:00",
    "astronomical_twilight_begin": "2026-06-12T10:59:04+00:00",
    "astronomical_twilight_end": "2026-06-12T06:13:07+00:00",
    "polar_condition": null
  },
  "status": "OK"
}

						

							day_length is measured in seconds.
							polar_condition is always included: it is
							null during a normal sunrise and sunset cycle,
							POLAR_DAY when the sun does not set, or
							POLAR_NIGHT when the sun does not rise. In a polar
							condition, sunrise, sunset, and
							day_length are null. Twilight times are
							calculated independently and may be a timestamp or
							null.
						

					


					

						
errors

						

							Invalid or missing parameters return HTTP 400 with a
							JSON error.
						

						
{
  "status": "ERROR",
  "error": "missing required query parameter: lat"
}
